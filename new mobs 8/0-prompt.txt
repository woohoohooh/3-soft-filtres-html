<script>

const canvas=document.getElementById('gameCanvas');

const ctx=canvas.getContext('2d');



canvas.width=window.innerWidth;

canvas.height=window.innerHeight;

canvas.focus();



let combo=0,streak=0,health=100;

let particles=[],enemies=[],starField=[],playerBolts=[];

let autoFireCooldown=0;

const keys={w:false,a:false,s:false,d:false};



const player={x:canvas.width/2,y:canvas.height-100,size:38,rotation:0,velocity:{x:0,y:0},speed:5,coreGlow:0,ringPhase:0};



class Particle{

constructor(x,y,color){this.x=x;this.y=y;this.color=color;this.size=Math.random()*6+2;this.speedX=Math.random()*5-2.5;this.speedY=Math.random()*5-2.5;this.life=.7+Math.random()*.3;this.decay=.02+Math.random()*.03}

update(){this.x+=this.speedX;this.y+=this.speedY;this.life-=this.decay;this.size*=.98}

draw(){ctx.save();ctx.globalAlpha=this.life;ctx.fillStyle=this.color;ctx.beginPath();ctx.arc(this.x,this.y,this.size,0,Math.PI*2);ctx.fill();ctx.restore()}

}



class PlayerBolt{

constructor(x,y,targetX,targetY){

this.x=x;this.y=y;

this.targetX=targetX;this.targetY=targetY;

this.age=0;this.life=1;

this.trail=[];

for(let i=0;i<8;i++)this.trail.push({x:this.x,y:this.y});

}

update(){

this.age++;

if(this.age>28)this.life-=.06;

if(!enemies.length)return;

let closest=null,closestDist=Infinity;

for(let e of enemies){

if(e.isDead)continue;

let dx=e.x-this.x,dy=e.y-this.y;

let d=Math.sqrt(dx*dx+dy*dy);

if(d<closestDist){closestDist=d;closest=e;}

}

if(closest){

this.targetX=closest.x;

this.targetY=closest.y;

}

let dx=this.targetX-this.x;

let dy=this.targetY-this.y;

let d=Math.sqrt(dx*dx+dy*dy);

if(d>1){

this.x+=dx*0.3;

this.y+=dy*0.3;

}

this.trail.shift();

this.trail.push({x:this.x,y:this.y});

}

draw(){

ctx.save();

ctx.globalAlpha=this.life;

ctx.shadowBlur=35;

ctx.shadowColor='#b0c4ff';

ctx.strokeStyle='#b0c4ff';

ctx.lineWidth=3;

ctx.beginPath();

ctx.moveTo(this.trail[0].x,this.trail[0].y);

for(let i=1;i<this.trail.length;i++)ctx.lineTo(this.trail[i].x,this.trail[i].y);

ctx.stroke();

ctx.shadowBlur=45;

ctx.lineWidth=1.8;

ctx.strokeStyle='#ffffff';

ctx.beginPath();

ctx.moveTo(this.trail[2].x,this.trail[2].y);

for(let i=3;i<this.trail.length;i++)ctx.lineTo(this.trail[i].x,this.trail[i].y);

ctx.stroke();

ctx.restore();

}

}



let bossSpawned=false;



class Enemy{

constructor(type){

this.type=type || Math.floor(Math.random()*3);

this.size=28;

this.speed=0.7;

this.health=35;

this.maxHealth=35;

this.rotation=0;

this.pulse=Math.random()*Math.PI*2;

this.hitCooldown=0;

this.isDead=false;

this.nodes=[];

this.baseRadius=this.size;

if(this.type===0){

this.size=32; this.speed=0.55; this.health=50; this.maxHealth=50;

this.structure='ring';

}else if(this.type===1){

this.size=24; this.speed=0.9; this.health=30; this.maxHealth=30;

this.structure='spike';

}else{

this.size=40; this.speed=0.45; this.health=70; this.maxHealth=70;

this.structure='crystal';

}

for(let i=0;i<5;i++)this.nodes.push({angle:(i/5)*Math.PI*2,offset:Math.random()*0.5+0.8});

const side=Math.floor(Math.random()*3);

if(side===0){this.x=Math.random()*canvas.width;this.y=-60}

if(side===1){this.x=-60;this.y=Math.random()*canvas.height*0.6}

if(side===2){this.x=canvas.width+60;this.y=Math.random()*canvas.height*0.6}

}

update(){

if(this.isDead)return;

const dx=player.x-this.x,dy=player.y-this.y;

const d=Math.sqrt(dx*dx+dy*dy);

if(d>0){this.x+=(dx/d)*this.speed;this.y+=(dy/d)*this.speed}

this.rotation+=0.005;this.pulse+=0.04;

if(this.hitCooldown>0)this.hitCooldown--;

if(d<player.size+this.size+15 && this.hitCooldown<=0){

health=Math.max(0,health-8);

streak=0;

this.hitCooldown=30;

}

}

draw(){

if(this.isDead)return;

ctx.save();

ctx.translate(this.x,this.y);

ctx.rotate(this.rotation);

let glowPulse=Math.sin(this.pulse)*0.2+0.8;

ctx.shadowBlur=30;

ctx.shadowColor='rgba(140,180,255,0.6)';

ctx.strokeStyle='rgba(180,210,255,0.7)';

ctx.lineWidth=1.8;

ctx.fillStyle='rgba(10,12,22,0.9)';

if(this.structure==='ring'){

ctx.beginPath();

ctx.arc(0,0,this.size*0.8,0,Math.PI*2);

ctx.fill();

ctx.stroke();

ctx.shadowBlur=40;

ctx.strokeStyle='rgba(200,220,255,0.9)';

ctx.lineWidth=1.2;

ctx.beginPath();

ctx.arc(0,0,this.size*0.4,0,Math.PI*2);

ctx.stroke();

for(let i=0;i<4;i++){

let a=i*Math.PI/2;

ctx.beginPath();

ctx.moveTo(Math.cos(a)*this.size*0.7,Math.sin(a)*this.size*0.7);

ctx.lineTo(Math.cos(a)*this.size*1.1,Math.sin(a)*this.size*1.1);

ctx.stroke();

}

}else if(this.structure==='spike'){

ctx.beginPath();

for(let i=0;i<6;i++){

let a=i*Math.PI/3;

let r=this.size*(i%2?1.0:0.6);

if(i===0)ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r);

else ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r);

}

ctx.closePath();

ctx.fill();

ctx.stroke();

ctx.shadowBlur=45;

ctx.strokeStyle='rgba(210,230,255,0.8)';

ctx.beginPath();

ctx.arc(0,0,this.size*0.3,0,Math.PI*2);

ctx.stroke();

}else{

ctx.beginPath();

for(let i=0;i<5;i++){

let n=this.nodes[i];

let a=n.angle+this.pulse*0.5;

let r=this.size*n.offset;

if(i===0)ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r);

else ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r);

}

ctx.closePath();

ctx.fill();

ctx.stroke();

ctx.shadowBlur=50;

ctx.fillStyle='rgba(40,60,100,0.3)';

ctx.beginPath();

ctx.arc(0,0,this.size*0.3,0,Math.PI*2);

ctx.fill();

}

if(this.health<this.maxHealth){

ctx.save();

ctx.rotate(-this.rotation);

ctx.translate(0,-this.size-12);

ctx.fillStyle='rgba(30,30,40,.8)';

ctx.fillRect(-this.size,0,this.size*2,3);

ctx.fillStyle='rgba(200,220,255,.95)';

ctx.fillRect(-this.size,0,(this.health/this.maxHealth)*this.size*2,3);

ctx.restore();

}

ctx.restore();

}

hit(dmg){

if(this.isDead)return false;

this.health-=dmg;

for(let i=0;i<8;i++)particles.push(new Particle(this.x+Math.random()*16-8,this.y+Math.random()*16-8,'#ccdfff'));

if(this.health<=0){

this.isDead=true;

streak++;

for(let i=0;i<20;i++)particles.push(new Particle(this.x+Math.random()*24-12,this.y+Math.random()*24-12,i<8?'#ffffff':'#aac0ff'));

const idx=enemies.indexOf(this);

if(idx>-1){

enemies.splice(idx,1);

if(this.type===3){bossSpawned=false;}

else{

if(enemies.length<6)enemies.push(new Enemy(Math.floor(Math.random()*3)));

}

}

return true;

}

return false;

}

}



function drawPlayer(){

player.coreGlow+=0.2;

player.ringPhase+=0.1;

ctx.save();

ctx.translate(player.x,player.y);

ctx.rotate(player.rotation);

let corePulse=0.5+Math.sin(player.coreGlow)*0.2;

ctx.shadowBlur=40;

ctx.shadowColor='rgba(160,200,255,0.9)';

ctx.strokeStyle='rgba(220,240,255,1)';

ctx.lineWidth=2.5;

ctx.fillStyle='rgba(5,10,18,0.85)';

ctx.beginPath();

ctx.moveTo(0,-player.size*1.3);

ctx.lineTo(player.size*0.8,player.size*0.2);

ctx.lineTo(player.size*0.9,player.size*0.6);

ctx.lineTo(player.size*0.4,player.size*0.4);

ctx.lineTo(0,player.size*0.7);

ctx.lineTo(-player.size*0.4,player.size*0.4);

ctx.lineTo(-player.size*0.9,player.size*0.6);

ctx.lineTo(-player.size*0.8,player.size*0.2);

ctx.closePath();

ctx.fill();

ctx.stroke();

ctx.shadowBlur=50;

ctx.shadowColor='#ffffff';

ctx.fillStyle='rgba(200,220,255,0.7)';

ctx.beginPath();

ctx.arc(0,-player.size*0.5,player.size*0.18,0,Math.PI*2);

ctx.fill();

ctx.shadowBlur=60;

ctx.fillStyle='#ffffff';

ctx.beginPath();

ctx.arc(0,-player.size*0.5,player.size*0.08,0,Math.PI*2);

ctx.fill();

ctx.restore();

}



function createStarField(){

if(Math.random()<0.3){

starField.push({x:Math.random()*canvas.width,y:-5,speed:1+Math.random()*3,size:Math.random()*2+1});

}

for(let i=starField.length-1;i>=0;i--){

let s=starField[i];

s.y+=s.speed;

if(s.y>canvas.height+10)starField.splice(i,1);

}

}



function autoAttack(){

let nearest=null,nd=Infinity;

for(const e of enemies){if(e.isDead)continue;const dx=e.x-player.x,dy=e.y-player.y;const d=Math.sqrt(dx*dx+dy*dy);if(d<nd){nd=d;nearest=e}}

if(nearest){

player.rotation=Math.atan2(nearest.y-player.y,nearest.x-player.x)*0.3;

if(nd<600){

for(let i=0;i<2;i++){

let offsetX=(Math.random()-0.5)*18;

let offsetY=(Math.random()-0.5)*18;

let bolt=new PlayerBolt(player.x+offsetX,player.y+offsetY,nearest.x+Math.random()*20-10,nearest.y+Math.random()*20-10);

playerBolts.push(bolt);

}

nearest.hit(16);

combo++;

health=Math.min(100,health+2);

}

}

}



function spawnInitialEnemies(){

for(let i=0;i<4;i++)enemies.push(new Enemy(Math.floor(Math.random()*3)));

if(!bossSpawned){enemies.push(new Enemy(3));bossSpawned=true;}

}



function update(){

player.velocity.x=0;player.velocity.y=0;

if(keys.w)player.velocity.y=-player.speed;

if(keys.s)player.velocity.y=player.speed;

if(keys.a)player.velocity.x=-player.speed;

if(keys.d)player.velocity.x=player.speed;

player.x=Math.min(canvas.width-player.size*.8,Math.max(player.size*.8,player.x+player.velocity.x));

player.y=Math.min(canvas.height-player.size*.8,Math.max(player.size*.8,player.y+player.velocity.y));

autoFireCooldown--;

if(autoFireCooldown<=0){autoAttack();autoFireCooldown=16}

for(const e of enemies)e.update();

for(let i=playerBolts.length-1;i>=0;i--){

playerBolts[i].update();

if(playerBolts[i].life<=0.1||playerBolts[i].age>36)playerBolts.splice(i,1);

}

for(let i=particles.length-1;i>=0;i--){particles[i].update();if(particles[i].life<=.01)particles.splice(i,1)}

createStarField();

}



function draw(){

ctx.clearRect(0,0,canvas.width,canvas.height);

for(let s of starField){

ctx.save();

ctx.globalAlpha=0.5+Math.random()*0.3;

ctx.fillStyle='#aac8ff';

ctx.shadowBlur=8;

ctx.shadowColor='#aac8ff';

ctx.beginPath();

ctx.arc(s.x,s.y,s.size,0,Math.PI*2);

ctx.fill();

ctx.restore();

}

particles.forEach(p=>p.draw());

playerBolts.forEach(b=>b.draw());

enemies.forEach(e=>e.draw());

drawPlayer();

}



function animate(){update();draw();requestAnimationFrame(animate)}



function handleResize(){

canvas.width=window.innerWidth;canvas.height=window.innerHeight;

player.x=Math.min(canvas.width-player.size*.8,Math.max(player.size*.8,player.x));

player.y=Math.min(canvas.height-player.size*.8,Math.max(player.size*.8,player.y));

}



window.addEventListener('keydown',e=>{const k=e.key.toLowerCase();if(keys[k]!==undefined){keys[k]=true;e.preventDefault()}});

window.addEventListener('keyup',e=>{const k=e.key.toLowerCase();if(keys[k]!==undefined){keys[k]=false;e.preventDefault()}});

window.addEventListener('resize',handleResize);



spawnInitialEnemies();

animate();

</script>
