seg.offset=Math.sin(this.age*0.8+i*0.5)*this.life*5;

}

}

let bossSpawned=false;

class Enemy{

constructor(type){

this.type=type || Math.floor(Math.random()*4);

this.size=30;

this.speed=0.8;

this.health=40;

this.maxHealth=40;

this.rotation=Math.random()*Math.PI*2;

this.rotSpeed=0.01;

this.hitCooldown=0;

this.isDead=false;

this.wingAngle=0;

this.glowIntensity=0;

if(this.type===0){

this.size=35;

this.speed=0.6;

this.health=50;

this.maxHealth=50;

this.wingCount=3;

}

else if(this.type===1){

this.size=28;

this.speed=0.9;

this.health=35;

this.maxHealth=35;

this.wingCount=4;

}

else if(this.type===2){

this.size=42;

this.speed=0.5;

this.health=70;

this.maxHealth=70;

this.wingCount=2;

}

else if(this.type===3){

this.size=60;

this.speed=1.5;

this.health=200;

this.maxHealth=200;

this.wingCount=6;

bossSpawned=true;

}

const side=Math.floor(Math.random()*3);

if(side===0){this.x=Math.random()*canvas.width;this.y=-100}

if(side===1){this.x=-100;this.y=Math.random()*canvas.height*0.6}

if(side===2){this.x=canvas.width+100;this.y=Math.random()*canvas.height*0.6}

}

update(){

if(this.isDead)return;

const dx=player.x-this.x,dy=player.y-this.y;

const d=Math.sqrt(dx*dx+dy*dy);

if(d>0){this.x+=(dx/d)*this.speed;this.y+=(dy/d)*this.speed}

this.rotation+=0.008;

this.wingAngle+=0.05;

this.glowIntensity=0.3+Math.sin(Date.now()*0.003+this.x)*0.2;

if(this.hitCooldown>0)this.hitCooldown--;

if(d<player.size+this.size+20 && this.hitCooldown<=0){

health=Math.max(0,health-8);

streak=0;

this.hitCooldown=30;

}

}

draw(){

if(this.isDead)return;

ctx.save();

ctx.translate(this.x,this.y);

ctx.rotate(this.rotation);

ctx.shadowBlur=40;

ctx.shadowColor='rgba(180,220,255,0.9)';

ctx.strokeStyle='rgba(220,240,255,0.8)';

ctx.lineWidth=2.5;

ctx.fillStyle='rgba(10,10,20,0.95)';

ctx.beginPath();

ctx.arc(0,0,this.size*0.7,0,Math.PI*2);

ctx.fill();

ctx.stroke();

for(let w=0;w<this.wingCount;w++){

let angle=w*(Math.PI*2/this.wingCount)+Math.sin(this.wingAngle)*0.2;

ctx.save();

ctx.rotate(angle);

ctx.shadowBlur=50;

ctx.shadowColor='rgba(200,220,255,0.8)';

ctx.strokeStyle='rgba(210,230,255,0.7)';

ctx.lineWidth=2;

ctx.beginPath();

ctx.moveTo(this.size*0.5,0);

ctx.lineTo(this.size*1.4,-this.size*0.6);

ctx.lineTo(this.size*1.8,0);

ctx.lineTo(this.size*1.4,this.size*0.6);

ctx.closePath();

ctx.stroke();

ctx.fillStyle='rgba(20,20,30,0.8)';

ctx.fill();

ctx.restore();

}

ctx.restore();

ctx.save();

ctx.translate(this.x,this.y);

ctx.shadowBlur=60;

ctx.shadowColor='rgba(255,255,255,0.7)';

ctx.fillStyle='rgba(200,220,240,0.2)';

ctx.beginPath();

ctx.arc(0,0,this.size*1.1,0,Math.PI*2);

ctx.fill();

ctx.restore();

if(this.health<this.maxHealth){

ctx.save();

ctx.translate(this.x,this.y-this.size-15);

ctx.fillStyle='rgba(40,40,60,.7)';

ctx.fillRect(-this.size,0,this.size*2,5);

ctx.fillStyle='rgba(220,240,255,.9)';

ctx.fillRect(-this.size,0,(this.health/this.maxHealth)*this.size*2,5);

ctx.restore();

}

}

hit(dmg){

if(this.isDead)return false;

this.health-=dmg;

for(let i=0;i<8;i++)particles.push(new Particle(this.x,this.y,'rgba(255,255,255,0.9)'));

if(this.health<=0){

this.isDead=true;

streak++;

for(let i=0;i<25;i++)particles.push(new Particle(this.x,this.y,i<10?'rgba(255,255,255,1)':'rgba(200,220,255,0.8)'));

const idx=enemies.indexOf(this);

if(idx>-1){

enemies.splice(idx,1);

if(this.type===3){

bossSpawned=false;

}

else{

let newEnemy=null;

if(this.type===0)newEnemy=new Enemy(0);

else if(this.type===1)newEnemy=new Enemy(1);

else if(this.type===2)newEnemy=new Enemy(2);

if(newEnemy)enemies.push(newEnemy);

}

}

return true;

}

return false;

}

}

function createSakuraPetal(){

if(Math.random()>.2)return;

sakuraPetals.push({x:Math.random()*canvas.width,y:-20,speedX:Math.random()*1-.5,speedY:.5+Math.random()*1,rotation:Math.random()*Math.PI*2,rotationSpeed:Math.random()*.02-.01,size:5+Math.random()*8,life:1});

}

function autoAttack(){

let nearest=null,nd=Infinity;

for(const e of enemies){if(e.isDead)continue;const dx=e.x-player.x,dy=e.y-player.y;const d=Math.sqrt(dx*dx+dy*dy);if(d<nd){nd=d;nearest=e}}

if(nearest){

player.rotation=Math.atan2(nearest.y-player.y,nearest.x-player.x);

if(nd<550){

for(let i=0;i<2;i++){

let offsetX=(Math.random()-0.5)*20;

let offsetY=(Math.random()-0.5)*20;

let bolt=new PlayerBolt(player.x+offsetX,player.y+offsetY,nearest.x,nearest.y);

playerBolts.push(bolt);

}

nearest.hit(18);

combo++;

health=Math.min(100,health+3);

}

}

}

}

function update(){

player.velocity.x=0;player.velocity.y=0;

if(keys.w)player.velocity.y=-player.speed;

if(keys.s)player.velocity.y=player.speed;

if(keys.a)player.velocity.x=-player.speed;

if(keys.d)player.velocity.x=player.speed;

player.x=Math.min(canvas.width-player.size*.8,Math.max(player.size*.8,player.x+player.velocity.x));

player.y=Math.min(canvas.height-player.size*.8,Math.max(player.size*.8,player.y+player.velocity.y));

autoFireCooldown--;

if(autoFireCooldown<=0){autoAttack();autoFireCooldown=15}

for(const e of enemies)e.update();

for(let i=playerBolts.length-1;i>=0;i--){

playerBolts[i].update();

if(playerBolts[i].life<=0.1||playerBolts[i].age>40)playerBolts.splice(i,1);

}

for(let i=particles.length-1;i>=0;i--){particles[i].update();if(particles[i].life<=.01)particles.splice(i,1)}

for(let i=sakuraPetals.length-1;i>=0;i--){

const p=sakuraPetals[i];

p.x+=p.speedX;p.y+=p.speedY;p.rotation+=p.rotationSpeed;p.life-=.002;

if(p.y>canvas.height||p.life<=0)sakuraPetals.splice(i,1);

}

createSakuraPetal();

}

ctx.beginPath();ctx.moveTo(0,-p.size);ctx.lineTo(p.size*.8,p.size*.6);ctx.lineTo(-p.size*.8,p.size*.6);ctx.closePath();ctx.fill();ctx.restore();

}

particles.forEach(p=>p.draw());

playerBolts.forEach(b=>b.draw());

enemies.forEach(e=>e.draw());

drawPlayer();

}
